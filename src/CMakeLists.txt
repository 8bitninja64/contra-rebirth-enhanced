cmake_minimum_required(VERSION 3.16)
project(chocorecomp CXX)
if(WIN32)
  enable_language(RC)
endif()
set(CMAKE_CXX_STANDARD 17)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE RelWithDebInfo)
endif()

# The runtime's HLE registry (runtime/hle_registry.cpp) from hle.txt: one HLE(Name) per
# replacement that is not a "Name = value" stub. The runtime builds without any game code.
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/hle.txt)
file(STRINGS ${CMAKE_CURRENT_SOURCE_DIR}/hle.txt HLE_LINES)
set(HLE_INC "")
foreach(line IN LISTS HLE_LINES)
  string(REGEX REPLACE "#.*$" "" line "${line}")
  string(STRIP "${line}" line)
  if(line AND NOT line MATCHES "=")
    string(APPEND HLE_INC "HLE(${line})\n")
  endif()
endforeach()
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/hle_registry.inc.tmp "${HLE_INC}")
configure_file(${CMAKE_CURRENT_BINARY_DIR}/hle_registry.inc.tmp ${CMAKE_CURRENT_BINARY_DIR}/hle_registry.inc COPYONLY)

# ---------------------------------------------------------------- runtime (no game code)
file(GLOB RT_SRC runtime/*.cpp runtime/enhancements/*.cpp)
list(REMOVE_ITEM RT_SRC ${CMAKE_CURRENT_SOURCE_DIR}/runtime/module_rt.cpp)
add_executable(contrarebirth ${RT_SRC})
target_include_directories(contrarebirth PRIVATE runtime runtime/enhancements ${CMAKE_CURRENT_BINARY_DIR})
# Dear ImGui for the F1 menu (windowed builds), fetched by tools/fetch_imgui.py
set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/build-deps/imgui")
if(WIN32 OR USE_SDL)
  if(NOT EXISTS "${IMGUI_DIR}/imgui.h")
    message(FATAL_ERROR "Dear ImGui is missing: run python tools/fetch_imgui.py")
  endif()
  target_sources(contrarebirth PRIVATE ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp
                 ${IMGUI_DIR}/imgui_widgets.cpp ${IMGUI_DIR}/backends/imgui_impl_sdl2.cpp ${IMGUI_DIR}/backends/imgui_impl_opengl3.cpp)
  target_include_directories(contrarebirth PRIVATE ${IMGUI_DIR})
endif()
if(WIN32)
  target_include_directories(contrarebirth PRIVATE ${SDL2_DIR}/include/SDL2)   # ImGui's backend includes <SDL.h>
  # SDL2 development package (SDL2-devel-*-mingw) location
  set(SDL2_DIR "" CACHE PATH "SDL2 mingw dev package root (x86_64-w64-mingw32 subfolder)")
  target_include_directories(contrarebirth PRIVATE runtime/gl ${SDL2_DIR}/include)
  target_compile_definitions(contrarebirth PRIVATE SDL_MAIN_HANDLED)
  target_compile_options(contrarebirth PRIVATE -O2 -fno-strict-aliasing -w -g0)
  target_link_directories(contrarebirth PRIVATE ${SDL2_DIR}/lib)
  target_sources(contrarebirth PRIVATE runtime/app.rc)
  target_link_libraries(contrarebirth PRIVATE SDL2 winmm -static-libgcc -static-libstdc++ -Wl,-Bstatic -lwinpthread -Wl,-Bdynamic)
elseif(USE_SDL)
  target_compile_definitions(contrarebirth PRIVATE USE_SDL)
  target_include_directories(contrarebirth PRIVATE /usr/include/SDL2)
  target_compile_options(contrarebirth PRIVATE -O2 -fno-strict-aliasing -w -g0)
  target_link_libraries(contrarebirth PRIVATE SDL2 GL pthread dl)
else()
  target_compile_options(contrarebirth PRIVATE -O2 -fno-strict-aliasing -w -g0)
  target_link_libraries(contrarebirth PRIVATE EGL GL pthread dl)
endif()

# ---------------------------------------------------------------- game module (from main.dol)
# Generated by recomp.py into src/gen; on a player's machine tools/make_module builds the same
# files with a bundled compiler instead (runtime/module_abi.h).
file(GLOB GEN_SRC gen/*.cpp)
if(GEN_SRC)
  add_library(contrarebirth_game SHARED ${GEN_SRC} runtime/module_rt.cpp)
  set_target_properties(contrarebirth_game PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
  target_include_directories(contrarebirth_game PRIVATE runtime gen)
  target_compile_definitions(contrarebirth_game PRIVATE CONTRA_MODULE=1)
  target_compile_options(contrarebirth_game PRIVATE -O2 -fno-strict-aliasing -w -g0)
  if(WIN32)
    target_link_libraries(contrarebirth_game PRIVATE -static-libgcc -static-libstdc++ -Wl,-Bstatic -lwinpthread -Wl,-Bdynamic)
  endif()
  add_dependencies(contrarebirth contrarebirth_game)
else()
  message(STATUS "src/gen is empty: building the runtime only (the game module is made from the player's WAD)")
endif()
